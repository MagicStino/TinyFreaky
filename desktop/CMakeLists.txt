cmake_minimum_required(VERSION 3.25)
project(Tinyfreaky VERSION 0.3.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT VST3_SDK_ROOT)
 message(FATAL_ERROR "Pass -DVST3_SDK_ROOT=/path/to/vst3sdk")
endif()
set(VSTGUI_STANDALONE OFF CACHE BOOL "" FORCE)
set(VSTGUI_TOOLS OFF CACHE BOOL "" FORCE)
set(VSTGUI_ENABLE_LIVE_EDITING OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_WAYLAND_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VSTGUI_SUPPORT ON CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
add_subdirectory("${VST3_SDK_ROOT}" sdk)
set(SDK_ROOT "${VST3_SDK_ROOT}")
set(public_sdk_SOURCE_DIR "${VST3_SDK_ROOT}/public.sdk")
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../src/plaits_sources.txt" PLAITS_RELATIVE)
set(PLAITS_SOURCES ../src/plaits_bridge.cpp)
foreach(SOURCE IN LISTS PLAITS_RELATIVE)
 list(APPEND PLAITS_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/../${SOURCE}")
endforeach()
find_package(Threads REQUIRED)
add_library(tinyfreaky-plaits STATIC ${PLAITS_SOURCES})
target_compile_definitions(tinyfreaky-plaits PRIVATE TEST)
target_include_directories(tinyfreaky-plaits PRIVATE ../third_party/eurorack ../src)
target_link_libraries(tinyfreaky-plaits PUBLIC Threads::Threads)
smtg_add_vst3plugin(Tinyfreaky processor.cpp controller.cpp factory.cpp editor.cpp ../src/engine.c ../src/parameters.inc)
target_include_directories(Tinyfreaky PRIVATE . ../src)
target_link_libraries(Tinyfreaky PRIVATE tinyfreaky-plaits sdk vstgui_support)
if(UNIX AND NOT APPLE)
 target_link_libraries(Tinyfreaky PRIVATE m)
endif()
smtg_target_configure_version_file(Tinyfreaky)
if(SMTG_MAC)
 smtg_target_set_bundle(Tinyfreaky BUNDLE_IDENTIFIER "com.magicstino.tinyfreaky" COMPANY_NAME "MagicStino")
endif()
option(TINYFREAKY_BUILD_TESTS "Build desktop host smoke test" ON)
if(TINYFREAKY_BUILD_TESTS)
 enable_testing()
 add_executable(tinyfreaky-desktop-smoke smoke.cpp processor.cpp ../src/engine.c "${VST3_SDK_ROOT}/public.sdk/source/common/memorystream.cpp")
 target_include_directories(tinyfreaky-desktop-smoke PRIVATE . ../src)
 target_link_libraries(tinyfreaky-desktop-smoke PRIVATE tinyfreaky-plaits sdk sdk_hosting vstgui_support)
 # The SDK base library uses CFString functions on macOS; executable targets
 # do not receive the framework linkage supplied by smtg_add_vst3plugin.
 if(APPLE)
  find_library(TINYFREAKY_COREFOUNDATION CoreFoundation REQUIRED)
  target_link_libraries(tinyfreaky-desktop-smoke PRIVATE "${TINYFREAKY_COREFOUNDATION}")
 endif()
 if(UNIX AND NOT APPLE)
  target_link_libraries(tinyfreaky-desktop-smoke PRIVATE m)
 endif()
 add_test(NAME desktop-smoke COMMAND tinyfreaky-desktop-smoke)
endif()

smtg_target_add_plugin_resources(Tinyfreaky RESOURCES resources/tinyfreaky.uidesc)

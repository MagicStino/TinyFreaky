cmake_minimum_required(VERSION 3.25)
project(Microfreeze VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT VST3_SDK_ROOT)
 message(FATAL_ERROR "Pass -DVST3_SDK_ROOT=/path/to/vst3sdk")
endif()
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
add_subdirectory("${VST3_SDK_ROOT}" sdk)
set(SDK_ROOT "${VST3_SDK_ROOT}")
set(public_sdk_SOURCE_DIR "${VST3_SDK_ROOT}/public.sdk")
smtg_add_vst3plugin(Microfreeze processor.cpp controller.cpp factory.cpp ../src/engine.c ../src/parameters.inc)
target_include_directories(Microfreeze PRIVATE . ../src)
target_link_libraries(Microfreeze PRIVATE sdk)
if(UNIX AND NOT APPLE)
 target_link_libraries(Microfreeze PRIVATE m)
endif()
smtg_target_configure_version_file(Microfreeze)
if(SMTG_MAC)
 smtg_target_set_bundle(Microfreeze BUNDLE_IDENTIFIER "com.magicstino.microfreeze" COMPANY_NAME "MagicStino")
endif()
option(MICROFREEZE_BUILD_TESTS "Build desktop host smoke test" ON)
if(MICROFREEZE_BUILD_TESTS)
 enable_testing()
 add_executable(microfreeze-desktop-smoke smoke.cpp processor.cpp ../src/engine.c "${VST3_SDK_ROOT}/public.sdk/source/common/memorystream.cpp")
 target_include_directories(microfreeze-desktop-smoke PRIVATE . ../src)
 target_link_libraries(microfreeze-desktop-smoke PRIVATE sdk sdk_hosting)
 if(UNIX AND NOT APPLE)
  target_link_libraries(microfreeze-desktop-smoke PRIVATE m)
 endif()
 add_test(NAME desktop-smoke COMMAND microfreeze-desktop-smoke)
endif()
